Запустить на ВМ как systemd сервис приложение https://github.com/anestesia001/simple-api
```
#создание директории
mkdir /opt/simple-api

#cоздание файлов приложения 
touch app2.py
touch requirements.txt

#оркужение и установка реквайментс
python3 -m venv venv
./venv/bin/pip install -r requirements.txt

Installing collected packages: markupsafe, itsdangerous, gunicorn, click, blinker, werkzeug, jinja2, Flask
Successfully installed Flask-3.1.3 blinker-1.9.0 click-8.5.0 gunicorn-26.2.0 itsdangerous-2.2.0 jinja2-3.1.6 markupsafe-3.0.3 werkzeug-3.1.


# юнит файл
[Unit]
Description=Simple Flask API Application
After=network.target

[Service]
Type=simple
User=root
WorkingDirectory=/opt/simple-api
ExecStart=/opt/simple-api/venv/bin/gunicorn --bind 0.0.0.0:8080 --pid /tmp/simple-api.pid app:app
Restart=on-failure
RestartSec=5

[Install]
WantedBy=multi-user.target


# запуск юнита
sudo systemctl daemon-reload
sudo systemctl enable simple-api
systemctl status simple-api
```

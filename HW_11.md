Запустить на ВМ как systemd сервис приложение на python https://github.com/anestesia001/apps_v2.2/blob/main/app2.py

```
#создание диерктории
mkdir /opt/app2

#cоздание файла приложение 
touch app2.py

#конфигурационный файл юнита:
sudo nano /etc/systemd/system/myapp.service

root@ubundu26:/opt/app2# nano /etc/systemd/system/app2.service

  GNU nano 8.7.1                       /etc/systemd/system/app2.service                                
[Unit]
Description=app2
After=network.target

[Service]
Type=simple
User=root
WorkingDirectory=/opt/app2
ExecStart=/opt/app2/venv/bin/python3 /opt/app2/app2.py
Restart=on-failure
RestartSec=5

[Install]
WantedBy=multi-user.target

# запуск юнита
sudo systemctl daemon-reload
sudo systemctl enable app2
systemctl status app2
```
